50 [[nodiscard]] inline double rational_precise(const double squared) noexcept
51 {
52 const double numerator{
53 squared *
54 (k_precise_p0 + squared *
55 (k_precise_p1 + squared *
56 (k_precise_p2 + squared *
57 (k_precise_p3 + squared *
58 (k_precise_p4 + squared * k_precise_p5)))))
59 };
60
61 const double denominator{
62 1.0 + squared *
63 (k_precise_q1 + squared *
64 (k_precise_q2 + squared *
65 (k_precise_q3 + squared * k_precise_q4)))
66 };
67
68 return numerator / denominator;
69 }

71 [[nodiscard]] inline double corrected_sqrt(const double value) noexcept
72 {
73 const double estimate{ static_cast<double>(sqrt_precise(static_cast<float>(value))) };
74
75 return estimate + (value - estimate * estimate) / (2.0 * estimate);
76 }

78 [[nodiscard]] inline float atan_kernel_fast(const float value) noexcept
79 {
80 const float squared{ value * value };
81 float polynomial{ k_atan_fast_c0 };
82 polynomial = polynomial * squared + k_atan_fast_c1;
83 polynomial = polynomial * squared + k_atan_fast_c2;
84 polynomial = polynomial * squared + k_atan_fast_c3;
85 polynomial = polynomial * squared + k_atan_fast_c4;
86 polynomial = polynomial * squared + k_atan_fast_c5;
87
88 return polynomial * value;
89 }

91 [[nodiscard]] inline double atan_kernel_precise(const double value) noexcept
92 {
93 const double squared{ value * value };
94 const double polynomial{
95 1.0 + squared *
96 (-1.0 / 3.0 + squared *
97 (1.0 / 5.0 + squared *
98 (-1.0 / 7.0 + squared *
99 (1.0 / 9.0 + squared *
100 (-1.0 / 11.0 + squared *
101 (1.0 / 13.0 + squared *
102 (-1.0 / 15.0 + squared *
103 (1.0 / 17.0 + squared * (-1.0 / 19.0)))))))))
104 };
105
106 return value * polynomial;
107 }

109 [[nodiscard]] inline float atan_positive_fast(const float value) noexcept

116 [[nodiscard]] inline float atan_positive_precise(const float value) noexcept
117 {
118 const double x{ value };
119 double base{ 0.0 };
120 double reduced{ x };
121
122 if (x > k_precise_tan_three_eighths_pi)
123 {
124 base = k_precise_half_pi_high;
125 reduced = -1.0 / x;
126 }
127 else if (x > k_precise_tan_eighth_pi)
128 {
129 base = k_precise_quarter_pi;
130 reduced = (x - 1.0) / (x + 1.0);
131 }
132
133 return static_cast<float>(base + atan_kernel_precise(reduced));
134 }

137 [[nodiscard]] inline float atan2_impl(const float y, const float x) noexcept
138 {
139 if (isnan(y)) return y;
140 if (isnan(x)) return x;
141
142 constexpr std::uint32_t magnitude_mask{ 0x7FFFFFFFu };
143 const std::uint32_t y_bits{ float_to_bits(y) };
144 const std::uint32_t x_bits{ float_to_bits(x) };
145 const std::uint32_t y_magnitude_bits{ y_bits & magnitude_mask };
146 const std::uint32_t x_magnitude_bits{ x_bits & magnitude_mask };
147 const bool y_negative{ y_bits >> 31u != 0u };
148 const bool x_negative{ x_bits >> 31u != 0u };
149
150 if (y_magnitude_bits == 0u)
151 return x_negative ? copysign(k_pi, y) : y;
152
153 if (x_magnitude_bits == 0u) return copysign(k_half_pi, y);
154
155 const bool y_infinite{ y_magnitude_bits == 0x7F800000u };
156 const bool x_infinite{ x_magnitude_bits == 0x7F800000u };
157
158 if (y_infinite && x_infinite)
159 {
160 const float angle{ x_negative ? 3.f * k_pi * .25f : k_pi * .25f };
161 return y_negative ? -angle : angle;
162 }
163
164 if (x_infinite)
165 return x_negative ? copysign(k_pi, y) : copysign(0.f, y);
166
167 if (y_infinite) return copysign(k_half_pi, y);
168
169 const float y_magnitude{ bits_to_float(y_magnitude_bits) };
170 const float x_magnitude{ bits_to_float(x_magnitude_bits) };
171 const bool steep{ y_magnitude > x_magnitude };
172 const float ratio{ steep ? x_magnitude / y_magnitude : y_magnitude / x_magnitude };
173 float angle;
174
175 if constexpr (Precise)
176 angle = atan_positive_precise(ratio);
177 else
178 angle = atan_kernel_fast(ratio);
179
180 if (steep)
181 angle = k_half_pi - angle;
182 if (x_negative)
183 angle = k_pi - angle;
184
185 return y_negative ? -angle : angle;
186 }

189 [[nodiscard]] inline float acos_fast(const float value) noexcept
190 {
191 if (isnan(value)) return value;
192 if (value > 1.f || value < -1.f) return quiet_nan();
193 if (value == 1.f) return 0.f;
194 if (value == -1.f) return k_pi;
195
196 const float magnitude{ bits_to_float(float_to_bits(value) & 0x7FFFFFFFu) };
197
198 if (magnitude == 0.f) return k_half_pi;
199
200 float polynomial{ inverse_trig_detail::k_fast_c0 };
201 polynomial = polynomial * magnitude + inverse_trig_detail::k_fast_c1;
202 polynomial = polynomial * magnitude + inverse_trig_detail::k_fast_c2;
203 polynomial = polynomial * magnitude + inverse_trig_detail::k_fast_c3;
204
205 const float approximation{ polynomial * sqrt_fast(1.f - magnitude) };
206
207 return signbit(value) ? k_pi - approximation : approximation;
208 }

210 [[nodiscard]] inline float acos_precise(const float value) noexcept
211 {
212 if (isnan(value)) return value;
213 if (value > 1.f || value < -1.f) return quiet_nan();
214 if (value == 1.f) return 0.f;
215 if (value == -1.f) return k_pi;
216
217 const double x{ value };
218 const std::uint32_t magnitude_bits{ float_to_bits(value) & 0x7FFFFFFFu };
219
220 if (magnitude_bits < 0x3F000000u)
221 {
222 if (magnitude_bits <= 0x32800000u) // |x| <= 2^-26
223 return static_cast<float>(inverse_trig_detail::k_precise_half_pi_high);
224
225 return static_cast<float>(inverse_trig_detail::k_precise_half_pi_high -
226 (x - (inverse_trig_detail::k_precise_half_pi_low -
227 x * inverse_trig_detail::rational_precise(x * x))));
228 }
229
230 const double squared{ (1.0 - (signbit(value) ? -x : x)) * 0.5 };
231 const double root{ inverse_trig_detail::corrected_sqrt(squared) };
232 const double approximation{ root + root * inverse_trig_detail::rational_precise(squared) };
233
234 if (signbit(value))
235 return static_cast<float>(2.0 * (inverse_trig_detail::k_precise_half_pi_high -
236 (approximation - inverse_trig_detail::k_precise_half_pi_low)));
237
238 return static_cast<float>(2.0 * approximation);
239 }

241 [[nodiscard]] inline float asin_fast(const float value) noexcept
242 {
243 if (isnan(value)) return value;
244 if (value > 1.f || value < -1.f) return quiet_nan();
245 if (value == 0.f) return value;
246 if (value == 1.f) return k_half_pi;
247 if (value == -1.f) return -k_half_pi;
248
249 const float magnitude{ bits_to_float(float_to_bits(value) & 0x7FFFFFFFu) };
250 float polynomial{ inverse_trig_detail::k_fast_c0 };
251 polynomial = polynomial * magnitude + inverse_trig_detail::k_fast_c1;
252 polynomial = polynomial * magnitude + inverse_trig_detail::k_fast_c2;
253 polynomial = polynomial * magnitude + inverse_trig_detail::k_fast_c3;
254 const float approximation{ k_half_pi - polynomial * sqrt_fast(1.f - magnitude) };
255
256 return signbit(value) ? -approximation : approximation;
257 }

259 [[nodiscard]] inline float asin_precise(const float value) noexcept
260 {
261 if (isnan(value)) return value;
262 if (value > 1.f || value < -1.f) return quiet_nan();
263 if (value == 0.f) return value;
264 if (value == 1.f) return k_half_pi;
265 if (value == -1.f) return -k_half_pi;
266
267 const double x{ value };
268 const double magnitude{ signbit(value) ? -x : x };
269 double approximation;
270
271 if (magnitude < 0.5)
272 {
273 approximation = magnitude + magnitude * inverse_trig_detail::rational_precise(magnitude * magnitude);
274 }
275 else
276 {
277 const double squared{ (1.0 - magnitude) * 0.5 };
278 const double root{ inverse_trig_detail::corrected_sqrt(squared) };
279 approximation = inverse_trig_detail::k_precise_half_pi_high -
280 2.0 * (root + root * inverse_trig_detail::rational_precise(squared));
281 }
282
283 const float result{ static_cast<float>(approximation) };
284
285 return signbit(value) ? -result : result;
286 }
