12 constexpr std::uint32_t k_sign_mask{ 0x80000000u };
13 constexpr std::uint32_t k_magnitude_mask{ 0x7FFFFFFFu };

15 // NOTE: Minimax-tuned for the two-step float Newton kernel below.
16 constexpr std::uint32_t k_sqrt_seed_bias{ 0x1FBAB000u };
17
18 // NOTE: Lomont-tuned inverse-square-root seed, followed by two Newton steps.
19 constexpr std::uint32_t k_rsqrt_seed_bias{ 0x5F375A86u };

21 constexpr float k_subnormal_scale{ 16777216.f }; // 2^24
22 constexpr float k_subnormal_result_scale{ 0.000244140625f }; // 2^-12

24 [[nodiscard]] inline float initial_estimate(const float value) noexcept
25 {
26 return bits_to_float((float_to_bits(value) >> 1u) + k_sqrt_seed_bias);
27 }

29 [[nodiscard]] inline float newton_step(const float value, const float estimate) noexcept
30 {
31 return 0.5f * (estimate + value / estimate);
32 }

34 [[nodiscard]] inline float sqrt_finite_positive_fast(const float value) noexcept
35 {
36 float estimate{ initial_estimate(value) };
37 estimate = newton_step(value, estimate);
38
39 return newton_step(value, estimate);
40 }

42 [[nodiscard]] inline float sqrt_finite_positive_precise(const float value) noexcept

49 [[nodiscard]] inline float rsqrt_newton_step(const float value, const float estimate) noexcept
50 {
51 return estimate * (1.5f - .5f * value * estimate * estimate);
52 }

54 [[nodiscard]] inline float rsqrt_finite_positive_fast(const float value) noexcept
55 {
56 float estimate{ bits_to_float(k_rsqrt_seed_bias - (float_to_bits(value) >> 1u)) };
57 estimate = rsqrt_newton_step(value, estimate);
58
59 return rsqrt_newton_step(value, estimate);
60 }

62 [[nodiscard]] inline float rsqrt_finite_positive_precise(const float value) noexcept
63 {
64 const double x{ value };
65 const double estimate{ rsqrt_finite_positive_fast(value) };
66
67 return static_cast<float>(estimate * (1.5 - .5 * x * estimate * estimate));

70 template<bool Precise>
71 [[nodiscard]] inline float sqrt_impl(const float value) noexcept
72 {
73 const std::uint32_t bits{ float_to_bits(value) };
74 const std::uint32_t magnitude{ bits & k_magnitude_mask };
75
76 // Returning the input preserves signed zero and NaN payloads.
77 if (magnitude == 0u || isnan(value)) return value;
78
79 if ((bits & k_sign_mask) != 0u) return quiet_nan();
80 if (isinf(value)) return value;
81
82 const bool subnormal{ exponent_bits(value) == 0u };
83 const float scaled_value{ subnormal ? value * k_subnormal_scale : value };
84
85 float result;
86
87 if constexpr (Precise)
88 result = sqrt_finite_positive_precise(scaled_value);
89 else
90 result = sqrt_finite_positive_fast(scaled_value);
91
92 return subnormal ? result * k_subnormal_result_scale : result;
93 }

95 template<bool Precise>
96 [[nodiscard]] inline float rsqrt_impl(const float value) noexcept
97 {
98 const std::uint32_t bits{ float_to_bits(value) };
99 const std::uint32_t magnitude{ bits & k_magnitude_mask };
100
101 if (isnan(value)) return value;
102 if (magnitude == 0u) return copysign(infinity(), value);
103 if ((bits & k_sign_mask) != 0u) return quiet_nan();
104 if (isinf(value)) return 0.f;
105
106 const bool subnormal{ exponent_bits(value) == 0u };
107 const float scaled_value{ subnormal ? value * k_subnormal_scale : value };
108 float result;
109
110 if constexpr (Precise)
111 result = rsqrt_finite_positive_precise(scaled_value);
112 else
113 result = rsqrt_finite_positive_fast(scaled_value);
114
115 return subnormal ? result / k_subnormal_result_scale : result;
116 }
