18 [[nodiscard]] constexpr float power_of_two(const int exponent) noexcept
19 {
20 if (exponent > 127) return infinity();
21
22 if (exponent >= -126)
23 return bits_to_float(static_cast<std::uint32_t>(exponent + 127) << 23u);
24
25 if (exponent >= -149)
26 return bits_to_float(1u << static_cast<unsigned>(exponent + 149));
27
28 return 0.f;
29 }

31 [[nodiscard]] inline float scale_by_power_of_two(const float value, const int exponent) noexcept
32 {
33 if (exponent == 128)
34 return value * 2.f * power_of_two(127);
35 if (exponent >= -126)
36 return value * power_of_two(exponent);
37
38 return value * power_of_two(exponent + 24) * k_subnormal_result_scale;
39 }

42 {

66 template<bool Precise>
67 [[nodiscard]] inline float exp2_impl(const float value) noexcept
68 {
69 if (isnan(value)) return value;
70 if (isinf(value)) return value > 0.f ? infinity() : 0.f;
71 if (value >= 128.f) return infinity();
72 if (value <= -150.f) return 0.f;
73
74 const int nearest_integer{ static_cast<int>(value + (value >= 0.f ? .5f : -.5f)) };
75 const float reduced{ value - static_cast<float>(nearest_integer) };
76
77 if (reduced == 0.f) return power_of_two(nearest_integer);
78
79 float approximation;
80
81 if constexpr (Precise)
82 approximation = static_cast<float>(exp_kernel_precise(
83 static_cast<double>(reduced) * k_ln_two));
84 else
85 approximation = exp_kernel_fast(reduced * static_cast<float>(k_ln_two));
86
87 return scale_by_power_of_two(approximation, nearest_integer);
88 }

91 [[nodiscard]] inline float exp_impl(const float value) noexcept
92 {
93 if (isnan(value)) return value;
94 if (isinf(value)) return value > 0.f ? infinity() : 0.f;
95 if (value >= 89.f) return infinity();
96 if (value <= -104.f) return 0.f;
97 if (value == 0.f) return 1.f;
98
99 const double scaled{ static_cast<double>(value) * k_inverse_ln_two };
100 const int nearest_integer{ static_cast<int>(scaled + (scaled >= 0.0 ? .5 : -.5)) };
101 const double reduced{ static_cast<double>(value) - static_cast<double>(nearest_integer) * k_ln_two };
102 float approximation;
103
104 if constexpr (Precise)
105 approximation = static_cast<float>(exp_kernel_precise(reduced));
106 else
107 approximation = exp_kernel_fast(static_cast<float>(reduced));
108
109 return scale_by_power_of_two(approximation, nearest_integer);

112 struct log_range_t
113 {
114 float mantissa;
115 int exponent;
116 };

118 [[nodiscard]] inline log_range_t reduce_log(const float value) noexcept
119 {
120 const bool subnormal{ exponent_bits(value) == 0u };
121 const float normalized{ subnormal ? value * k_subnormal_scale : value };
122 const std::uint32_t bits{ float_to_bits(normalized) };
123 int exponent{ static_cast<int>(bits >> 23u & 0xFFu) - 127 - (subnormal ? 24 : 0) };
124 float mantissa{ bits_to_float(bits & 0x007FFFFFu | 0x3F800000u) };
125
126 if (mantissa > k_sqrt_two)
127 {
128 mantissa *= .5f;
129 ++exponent;
130 }
131
132 return { mantissa, exponent };
133 }

135 [[nodiscard]] inline float log_kernel_fast(const float mantissa) noexcept
136 {
137 const float ratio{ (mantissa - 1.f) / (mantissa + 1.f) };
138 const float squared{ ratio * ratio };
139
140 const float series{
141 ratio * (1.f + squared *
142 (1.f / 3.f + squared *
143 (1.f / 5.f + squared *
144 (1.f / 7.f + squared * (1.f / 9.f)))))
145 };
146
147 return 2.f * series;
148 }

172 const double squared{ ratio * ratio };

194 [[nodiscard]] inline float log2_impl(const float value) noexcept
195 {
196 if (isnan(value)) return value;
197 if (value == 0.f) return -infinity();
198 if (signbit(value)) return quiet_nan();
199 if (isinf(value)) return infinity();

201 const log_range_t range{ reduce_log(value) };
202
