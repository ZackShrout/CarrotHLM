33 [[nodiscard]] constexpr std::uint32_t product_bit(const std::uint32_t (&product)[7], const int index) noexcept
34 {
35 if (index < 0 || index >= 224) return 0u;
36
37 return product[index / 32] >> index % 32 & 1u;
38 }

40 inline void multiply_two_over_pi(const std::uint32_t mantissa, std::uint32_t (&product)[7]) noexcept
41 {
42 std::uint64_t carry{ 0u };
43 for (int i{ 0 }; i < 6; ++i)
44 {
45 const std::uint64_t term{ static_cast<std::uint64_t>(k_two_over_pi[i]) * mantissa + carry };
46 product[i] = static_cast<std::uint32_t>(term);
47 carry = term >> 32u;
48 }
49 product[6] = static_cast<std::uint32_t>(carry);
50 }

52 [[nodiscard]] inline trig_range_t reduce_large_positive(const float value) noexcept
53 {
54 const std::uint32_t bits{ float_to_bits(value) };
55 const std::uint32_t mantissa{ bits & 0x007FFFFFu | 0x00800000u };
56 const int exponent{ static_cast<int>(bits >> 23u & 0xFFu) - 127 };
57 const int binary_point{ 215 - exponent };
58
59 std::uint32_t product[7]{ };
60 multiply_two_over_pi(mantissa, product);
61
62 // Keep the quotient mod 4 for quadrant selection and the leading
63 // 64 fractional bits for a reduced argument in [-pi/4, pi/4].
64 int quadrant{
65 static_cast<int>(product_bit(product, binary_point)) |
66 static_cast<int>(product_bit(product, binary_point + 1)) << 1
67 };
68
69 std::uint64_t fraction_bits{ 0u };
70
71 for (int i{ 1 }; i <= 64; ++i)
72 fraction_bits = fraction_bits << 1u | product_bit(product, binary_point - i);
73
74 double fraction{ static_cast<double>(fraction_bits) * k_two_pow_minus_64 };
75
76 if (product_bit(product, binary_point - 1) != 0u)
77 {
78 fraction -= 1.0;
79 quadrant = quadrant + 1 & 3;
80 }
81
82 return trig_range_t{ fraction * (k_half_pi_high + k_half_pi_tail), quadrant };
83 }

85 [[nodiscard]] inline trig_range_t reduce_medium_positive(const float value) noexcept
86 {
87 const double scaled{ static_cast<double>(value) * k_inverse_half_pi };
88 int quadrant_count{ static_cast<int>(scaled + 0.5) };
89 double reduced{
90 static_cast<double>(value) - quadrant_count * k_half_pi_high - quadrant_count * k_half_pi_tail
91 };
92
93 if (reduced < -k_quarter_pi)
94 {
95 --quadrant_count;
96 reduced += k_half_pi_high + k_half_pi_tail;
97 }
98 else if (reduced > k_quarter_pi)
99 {
100 ++quadrant_count;
101 reduced -= k_half_pi_high + k_half_pi_tail;
102 }
103
104 return trig_range_t{ reduced, quadrant_count & 3 };
105 }

108 [[nodiscard]] inline trig_range_t reduce_angle(const float value) noexcept
109 {
110 const bool negative{ signbit(value) };
111 const float magnitude{ bits_to_float(float_to_bits(value) & 0x7FFFFFFFu) };
112
113 trig_range_t result;
114
115 if (magnitude <= static_cast<float>(range_reduction_detail::k_quarter_pi))
116 result = trig_range_t{ magnitude, 0 };
117 else if (magnitude < range_reduction_detail::k_medium_reduction_limit)
118 result = range_reduction_detail::reduce_medium_positive(magnitude);
119 else
120 result = range_reduction_detail::reduce_large_positive(magnitude);
121
122 if (negative)
123 {
124 result.reduced = -result.reduced;
125 result.quadrant = 4 - result.quadrant & 3;
126 }
127
128 return result;
129 }
