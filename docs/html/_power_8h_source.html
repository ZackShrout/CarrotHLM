18 [[nodiscard]] inline exponent_class_t classify_exponent(const float value) noexcept
19 {
20 const float magnitude{ bits_to_float(float_to_bits(value) & 0x7FFFFFFFu) };
21
22 if (magnitude >= 16777216.f) // Every finite float >= 2^24 is an even integer.
23 return { true, false };
24
25 if (magnitude < 1.f) return { false, false };
26
27 const int integer{ static_cast<int>(magnitude) };
28
29 if (static_cast<float>(integer) != magnitude)
30 return { false, false };
31
32 return { true, (integer & 1) != 0 };
33 }

36 [[nodiscard]] inline float integer_power(const float base, const int exponent) noexcept
37 {
38 unsigned remaining{ static_cast<unsigned>(exponent < 0 ? -exponent : exponent) };
39
40 if constexpr (Precise)
41 {
42 double factor{ base };
43 double result{ 1.0 };
44
45 while (remaining != 0u)
46 {
47 if ((remaining & 1u) != 0u) result *= factor;
48
49 remaining >>= 1u;
50
51 if (remaining != 0u) factor *= factor;
52 }
53
54 return static_cast<float>(exponent < 0 ? 1.0 / result : result);
55 }
56 else

58 float factor{ base };
59 float result{ 1.f };
60

75 [[nodiscard]] inline float pow_impl(const float base, const float exponent) noexcept
76 {
77 if (exponent == 0.f || base == 1.f) return 1.f;
78 if (base == -1.f && isinf(exponent)) return 1.f;
79 if (isnan(base)) return base;
80 if (isnan(exponent)) return exponent;
81
82 const float base_magnitude{ bits_to_float(float_to_bits(base) & 0x7FFFFFFFu) };
83
84 if (isinf(exponent))
85 {
86 if (base_magnitude == 1.f) return 1.f;
87
88 const bool grows{ base_magnitude > 1.f };
89
90 return exponent > 0.f == grows ? infinity() : 0.f;
91 }
92
93 const exponent_class_t exponent_class{ classify_exponent(exponent) };
94
95 if (base_magnitude == 0.f)
96 {
97 const float magnitude{ exponent > 0.f ? 0.f : infinity() };
98
99 return signbit(base) && exponent_class.odd ? -magnitude : magnitude;
100 }
101
102 if (isinf(base))
103 {
104 const float magnitude{ exponent > 0.f ? infinity() : 0.f };
105
106 return signbit(base) && exponent_class.odd ? -magnitude : magnitude;
107 }
108
109 if (signbit(base) && !exponent_class.integral) return quiet_nan();
110
111 float magnitude;
112
113 if (exponent_class.integral && exponent >= -64.f && exponent <= 64.f)
114 {
115 magnitude = integer_power<Precise>(base_magnitude, static_cast<int>(exponent));
116 }
117 else
118 {
119 const double logarithm{ exponential_detail::log2_positive_finite<Precise>(base_magnitude) };
120 magnitude = exponential_detail::exp2_from_double<Precise>(logarithm * static_cast<double>(exponent));
121 }
122
123 return signbit(base) && exponent_class.odd ? -magnitude : magnitude;
124 }
