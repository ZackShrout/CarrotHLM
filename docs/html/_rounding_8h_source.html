35 [[nodiscard]] constexpr float trunc(const float value) noexcept
36 {
37 const std::uint32_t bits{ float_to_bits(value) };
38 const int exponent{ rounding_detail::unbiased_exponent(bits) };
39
40 if (exponent < 0) return rounding_detail::signed_zero(bits);
41 if (exponent >= 23) return value;
42
43 return bits_to_float(bits & ~rounding_detail::fractional_mask(exponent));
44 }

46 [[nodiscard]] constexpr float floor(const float value) noexcept
47 {
48 const std::uint32_t bits{ float_to_bits(value) };
49 const int exponent{ rounding_detail::unbiased_exponent(bits) };
50
51 if (exponent < 0)
52 {
53 if ((bits & rounding_detail::k_magnitude_mask) == 0u) return value;
54
55 return (bits & rounding_detail::k_sign_mask) == 0u ? 0.f : -1.f;
56 }
57
58 if (exponent >= 23) return value;
59
60 const std::uint32_t mask{ rounding_detail::fractional_mask(exponent) };
61
62 if ((bits & mask) == 0u) return value;
63
64 const std::uint32_t integral_bits{ bits & ~mask };
65
66 if ((bits & rounding_detail::k_sign_mask) == 0u)
67 return bits_to_float(integral_bits);
68
69 return bits_to_float(integral_bits) - 1.f;
70 }

99 [[nodiscard]] constexpr float round(const float value) noexcept
100 {
101 const std::uint32_t bits{ float_to_bits(value) };
102 const std::uint32_t magnitude{ bits & rounding_detail::k_magnitude_mask };
103 const int exponent{ rounding_detail::unbiased_exponent(bits) };
104
105 if (exponent < -1) return rounding_detail::signed_zero(bits);
106 if (exponent < 0) return copysign(1.f, value);
107 if (exponent >= 23) return value;
108
109 const std::uint32_t mask{ rounding_detail::fractional_mask(exponent) };
110
111 if ((magnitude & mask) == 0u) return value;
112
113 const std::uint32_t halfway{ 1u << (22 - exponent) };
114 const std::uint32_t rounded_magnitude{ magnitude + halfway & ~mask };
115
116 return bits_to_float(bits & rounding_detail::k_sign_mask | rounded_magnitude);
117 }

141 [[nodiscard]] constexpr float fmod(const float dividend, const float divisor) noexcept
142 {
143 const std::uint32_t dividend_bits{ float_to_bits(dividend) };
144 const std::uint32_t divisor_bits{ float_to_bits(divisor) };
145 std::uint32_t dividend_magnitude{ dividend_bits & rounding_detail::k_magnitude_mask };
146 std::uint32_t divisor_magnitude{ divisor_bits & rounding_detail::k_magnitude_mask };
147
148 if (isnan(dividend)) return dividend;
149 if (isnan(divisor)) return divisor;
150 if (divisor_magnitude == 0u || isinf(dividend)) return quiet_nan();
151 if (dividend_magnitude < divisor_magnitude) return dividend;
152 if (dividend_magnitude == divisor_magnitude) return rounding_detail::signed_zero(dividend_bits);
153
154 int dividend_exponent{ static_cast<int>(dividend_magnitude >> 23u & 0xFFu) };
155 int divisor_exponent{ static_cast<int>(divisor_magnitude >> 23u & 0xFFu) };
156 std::uint32_t shifted;
157
158 if (dividend_exponent == 0)
159 {
160 for (shifted = dividend_magnitude << 9u; shifted >> 31u == 0u; --dividend_exponent, shifted <<= 1u) {}
161
162 dividend_magnitude <<= static_cast<unsigned>(-dividend_exponent + 1);
163 }
164 else
165 {
166 dividend_magnitude = dividend_magnitude & rounding_detail::k_mantissa_mask | rounding_detail::k_hidden_bit;
167 }
168
169 if (divisor_exponent == 0)
170 {
171 for (shifted = divisor_magnitude << 9u; shifted >> 31u == 0u; --divisor_exponent, shifted <<= 1u) {}
172
173 divisor_magnitude <<= static_cast<unsigned>(-divisor_exponent + 1);
174 }
175 else
176 {
177 divisor_magnitude = divisor_magnitude & rounding_detail::k_mantissa_mask | rounding_detail::k_hidden_bit;
178 }
179
180 for (; dividend_exponent > divisor_exponent; --dividend_exponent)
181 {
182 shifted = dividend_magnitude - divisor_magnitude;
183
184 if (shifted >> 31u == 0u)
185 {
186 if (shifted == 0u) return rounding_detail::signed_zero(dividend_bits);
187
188 dividend_magnitude = shifted;
189 }
190
191 dividend_magnitude <<= 1u;
192 }
193
194 shifted = dividend_magnitude - divisor_magnitude;
195
196 if (shifted >> 31u == 0u)
197 {
198 if (shifted == 0u) return rounding_detail::signed_zero(dividend_bits);
199
200 dividend_magnitude = shifted;
201 }
202
203 while (dividend_magnitude >> 23u == 0u)
204 {
205 dividend_magnitude <<= 1u;
206 --dividend_exponent;
207 }
208
209 if (dividend_exponent > 0)
210 {
211 dividend_magnitude -= rounding_detail::k_hidden_bit;
212 dividend_magnitude |= static_cast<std::uint32_t>(dividend_exponent) << 23u;
213 }
214 else
215 {
216 dividend_magnitude >>= static_cast<unsigned>(-dividend_exponent + 1);
217 }
218
219 return bits_to_float(dividend_magnitude | dividend_bits & rounding_detail::k_sign_mask);
220 }

222 [[nodiscard]] constexpr float frac(const float value) noexcept
223 {
224 if (isnan(value)) return value;
225 if (isinf(value)) return quiet_nan();
226
227 return value - floor(value);
228 }
