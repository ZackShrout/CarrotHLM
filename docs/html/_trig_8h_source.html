41 [[nodiscard]] inline float sine_kernel_fast(const double value) noexcept
42 {
43 const float x{ static_cast<float>(value) };
44 const float squared{ x * x };
45
46 return x + x * squared * (k_fast_sin_1 + squared * (k_fast_sin_2 + squared * k_fast_sin_3));
47 }
48
49 [[nodiscard]] inline float cosine_kernel_fast(const double value) noexcept
50 {
51 const float x{ static_cast<float>(value) };
52 const float squared{ x * x };
53
54 return 1.f + squared * (k_fast_cos_0 + squared * (k_fast_cos_1 + squared * k_fast_cos_2));
55 }

57 [[nodiscard]] inline float sine_kernel_precise(const double value) noexcept
58 {
59 const double squared{ value * value };
60
61 return static_cast<float>(value + value * squared *
62 (k_precise_sin_1 + squared *
63 (k_precise_sin_2 + squared *
64 (k_precise_sin_3 + squared * k_precise_sin_4))));
65 }
66
67 [[nodiscard]] inline float cosine_kernel_precise(const double value) noexcept
68 {
69 const double squared{ value * value };
70
71 return static_cast<float>(1.0 + squared *
72 (k_precise_cos_0 + squared *
73 (k_precise_cos_1 + squared *
74 (k_precise_cos_2 + squared * k_precise_cos_3))));
75 }

95 template<bool Precise, bool Cosine>
96 [[nodiscard]] inline float trig_impl(const float value) noexcept
97 {
98 if (isnan(value))
99 return value;
100 if (isinf(value))
101 return quiet_nan();
102
103 const std::uint32_t magnitude_bits{ float_to_bits(value) & 0x7FFFFFFFu };
104 if (magnitude_bits < 0x39800000u) // |x| < 2^-12
105 {
106 if constexpr (Cosine)
107 return 1.f;
108 else
109 return value;
110 }
111
112 const trig_range_t range{ reduce_angle(value) };
113
114 if constexpr (Cosine)
115 {
116 switch (range.quadrant)
117 {
118 case 0: return evaluate_cosine<Precise>(range.reduced);
119 case 1: return -evaluate_sine<Precise>(range.reduced);
120 case 2: return -evaluate_cosine<Precise>(range.reduced);
121 default: return evaluate_sine<Precise>(range.reduced);
122 }
123 }
124 else
125 {
126 switch (range.quadrant)
127 {
128 case 0: return evaluate_sine<Precise>(range.reduced);
129 case 1: return evaluate_cosine<Precise>(range.reduced);
130 case 2: return -evaluate_sine<Precise>(range.reduced);
131 default: return -evaluate_cosine<Precise>(range.reduced);
132 }
133 }
134 }

136 template<bool Precise>
137 [[nodiscard]] inline float tangent_impl(const float value) noexcept
138 {
139 if (isnan(value))
140 return value;
141 if (isinf(value))
142 return quiet_nan();
143
144 const std::uint32_t magnitude_bits{ float_to_bits(value) & 0x7FFFFFFFu };
145
146 if (magnitude_bits < 0x39800000u) // |x| < 2^-12
147 return value;
148
149 const trig_range_t range{ reduce_angle(value) };
150 const float sine{ evaluate_sine<Precise>(range.reduced) };
151 const float cosine{ evaluate_cosine<Precise>(range.reduced) };
152
153 if ((range.quadrant & 1) == 0) return sine / cosine;
154
155 return -cosine / sine;
156 }

158 template<bool Precise>
159 [[nodiscard]] inline sin_cos_t sin_cos_impl(const float value) noexcept
160 {
161 if (isnan(value)) return { value, value };
162
163 if (isinf(value))
164 {
165 const float result{ quiet_nan() };
166
167 return { result, result };
168 }
169
170 const std::uint32_t magnitude_bits{ float_to_bits(value) & 0x7FFFFFFFu };
171
172 if (magnitude_bits < 0x39800000u) // |x| < 2^-12
173 return { value, 1.f };
174
175 const trig_range_t range{ reduce_angle(value) };
176 const float sine{ evaluate_sine<Precise>(range.reduced) };
177 const float cosine{ evaluate_cosine<Precise>(range.reduced) };
178
179 switch (range.quadrant)
180 {
181 case 0: return { sine, cosine };
182 case 1: return { cosine, -sine };
183 case 2: return { -sine, -cosine };
184 default: return { -cosine, sine };
185 }
