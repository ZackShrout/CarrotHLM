125 [[nodiscard]] inline float3 random_unit_vector3(rng32& rng) noexcept
126 {
127 while (true)
128 {
129 const float x{ rng.next_float(-1.f, 1.f) };
130 const float y{ rng.next_float(-1.f, 1.f) };
131 const float radius_squared{ x * x + y * y };
132
133 if (radius_squared <= 0.f || radius_squared >= 1.f) continue;
134
135 const float scale{ 2.f * sqrt(1.f - radius_squared) };
136
137 return { x * scale, y * scale, 1.f - 2.f * radius_squared };
138 }
139 }
