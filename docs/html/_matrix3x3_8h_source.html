211 inline float3x3 rotate_axis_angle(const float3 axis, const float rad) noexcept
212 {
213 const quat q{ quat_from_axis_angle(axis, rad) };
214
215 const float xx{ q.x * q.x };
216 const float yy{ q.y * q.y };
217 const float zz{ q.z * q.z };
218 const float xy{ q.x * q.y };
219 const float xz{ q.x * q.z };
220 const float yz{ q.y * q.z };
221 const float wx{ q.w * q.x };
222 const float wy{ q.w * q.y };
223 const float wz{ q.w * q.z };
224
225 return {
226 float3{ 1.f - 2.f * (yy + zz), 2.f * (xy - wz), 2.f * (xz + wy) },
227 float3{ 2.f * (xy + wz), 1.f - 2.f * (xx + zz), 2.f * (yz - wx) },
228 float3{ 2.f * (xz - wy), 2.f * (yz + wx), 1.f - 2.f * (xx + yy) }
229 };
