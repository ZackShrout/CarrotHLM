50 [[nodiscard]] constexpr float previous_float(const float value) noexcept
51 {
52 const std::uint32_t bits{ std::bit_cast<std::uint32_t>(value) };
53
54 if ((bits & 0x7FFFFFFFu) == 0u) return -std::numeric_limits<float>::denorm_min();
55
56 const std::uint32_t previous_bits{ (bits & 0x80000000u) == 0u ? bits - 1u : bits + 1u };
57
58 return std::bit_cast<float>(previous_bits);
59 }

61 [[nodiscard]] constexpr float next_float(
62 pcg32_t& engine,
63 const float min_inclusive,
64 const float max_exclusive) noexcept
65 {
66 assert(min_inclusive <= max_exclusive);
67 assert(is_finite(min_inclusive) && is_finite(max_exclusive));
68
69 const float unit{ unit_float(engine) };
70
71 if (min_inclusive == max_exclusive) return min_inclusive;
72
73 const double range{ static_cast<double>(max_exclusive) - static_cast<double>(min_inclusive) };
74 const float result{ static_cast<float>(static_cast<double>(min_inclusive) + range * unit) };
75
76 return result < max_exclusive ? result : previous_float(max_exclusive);
