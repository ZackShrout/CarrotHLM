234 inline float4 mul(const float4x4& m, const float4& v) noexcept
235 {
236 return v.x * m.columns[0] +
237 v.y * m.columns[1] +
238 v.z * m.columns[2] +
239 v.w * m.columns[3];
240 }
241
249 inline float4x4 mul(const float4x4& a, const float4x4& b) noexcept
250 {
251 float4x4 result;
252 result.columns[0] = mul(a, b.columns[0]);
253 result.columns[1] = mul(a, b.columns[1]);
254 result.columns[2] = mul(a, b.columns[2]);
255 result.columns[3] = mul(a, b.columns[3]);
256 return result;
257 }

332 inline float4x4 float4x4::rotate_axis_angle(float3 axis, const float rad) noexcept
333 {
334 axis = normalize(axis);
335 const float c{ cos(rad) };
336 const float s{ sin(rad) };
337 const float t{ 1.0f - c };
338
339 const float x{ axis.x };
340 const float y{ axis.y };
341 const float z{ axis.z };
342
343 return float4x4{
344 float4{ t * x * x + c, t * x * y + s * z, t * x * z - s * y, 0.f },
345 float4{ t * x * y - s * z, t * y * y + c, t * y * z + s * x, 0.f },
346 float4{ t * x * z + s * y, t * y * z - s * x, t * z * z + c, 0.f },
347 float4{ 0.f, 0.f, 0.f, 1.f }
348 };

379 inline float4x4 float4x4::perspective_lh(const float fov_y_rad, const float aspect, const float z_near,
380 const float z_far) noexcept
381 {
382 const float h{ 1.f / tan(fov_y_rad * .5f) };
383 const float w{ h / aspect };
384 const float d{ z_far / (z_far - z_near) };
385
386 return float4x4{
387 float4{ w, 0.f, 0.f, 0.f },
388 float4{ 0.f, h, 0.f, 0.f },
389 float4{ 0.f, 0.f, d, 1.f },
390 float4{ 0.f, 0.f, -d * z_near, 0.f }
391 };
