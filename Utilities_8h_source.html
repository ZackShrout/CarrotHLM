33
34 return static_cast<float>((static_cast<double>(value) - static_cast<double>(start)) /

55 [[nodiscard]] constexpr float remap(const float input_start, const float input_end,
56 const float output_start, const float output_end,
57 const float value) noexcept
58 {
59 const float factor{ inverse_lerp(input_start, input_end, value) };
60
61 if (factor == 0.f) return output_start;
62 if (factor == 1.f) return output_end;
63
64 return static_cast<float>(static_cast<double>(output_start) * (1.0 - static_cast<double>(factor)) +
65 static_cast<double>(output_end) * static_cast<double>(factor));

104 [[nodiscard]] constexpr float smoothstep(const float edge_start, const float edge_end,
105 const float value) noexcept
106 {
107 if (edge_start == edge_end) return step(edge_start, value);
108
109 const float t{ saturate(inverse_lerp(edge_start, edge_end, value)) };
110
111 if (t <= .5f) return t * t * (3.f - 2.f * t);
112
113 const float complement{ 1.f - t };
114
115 return 1.f - complement * complement * (3.f - 2.f * complement);
116 }

197 [[nodiscard]] constexpr float repeat(const float value, const float length) noexcept
198 {
199 if (length <= 0.f) return 0.f;
200 if (!isfinite(length)) return isnan(length) ? length : length - length; // length - length here produces NaN
201
202 const float remainder{ fmod(value, length) };
203
204 if (isnan(remainder)) return remainder;
205 if (remainder == 0.f) return 0.f;
206 if (remainder > 0.f) return remainder;
207
208 const float wrapped{ remainder + length };
209
210 return wrapped == length ? 0.f : wrapped;
211 }

231 [[nodiscard]] constexpr float wrap(const float value, const float bound_a, const float bound_b) noexcept
232 {
233 if (bound_a == bound_b) return bound_a;
234 if (isnan(value) || isnan(bound_a) || isnan(bound_b)) return value + bound_a + bound_b;
235
236 const float lower{ min(bound_a, bound_b) };
237 const float upper{ max(bound_a, bound_b) };
238
239 if (value >= lower && value < upper) return value;
240
241 const float width{ upper - lower };
242 const float value_phase{ repeat(value, width) };
243 const float lower_phase{ repeat(lower, width) };
244
245 if (isnan(value_phase) || isnan(lower_phase)) return value_phase + lower_phase;
246
247 const float offset{
248 value_phase >= lower_phase
249 ? value_phase - lower_phase
250 : width - (lower_phase - value_phase)
251 };
252 if (offset == width) return lower;
253
254 const float result{ lower + offset };
255 return result == upper ? lower : result;
256 }

323 [[nodiscard]] constexpr float ping_pong(const float value, const float length) noexcept
324 {
325 if (length <= 0.f) return 0.f;
326
327 const float magnitude{ abs(value) };
328 if (magnitude <= length) return magnitude;
329
330 const float period{ length + length };
331
332 if (isfinite(period))
333 {
334 const float phase{ repeat(magnitude, period) };
335
336 if (isnan(phase)) return phase;
337
338 return phase <= length ? phase : period - phase;
339 }
340
341 const float phase{ repeat(magnitude * .5f, length) };
342
343 if (isnan(phase)) return phase;
344
345 return phase <= length * .5f ? phase * 2.f : (length - phase) * 2.f;
346 }

372 [[nodiscard]] inline float exp_decay(const float current, const float target,
373 const float half_life, const float delta_time) noexcept
374 {
375 if (delta_time <= 0.f || current == target) return current;
376
377 if (isnan(current) || isnan(target) || isnan(half_life) || isnan(delta_time))
378 return current + target + half_life + delta_time;
379
380 if (half_life <= 0.f) return current;
381 if (isinf(delta_time)) return target;
382 if (isinf(half_life)) return current;
383
384 if (isinf(current) || isinf(target))
385 {
386 if (isinf(current) && isinf(target)) return current + target;
387
388 return isinf(current) ? current : target;
389 }
390
391 const float retained{ exp2_precise(-delta_time / half_life) };
392
393 if (retained == 0.f) return target;
394 if (retained == 1.f) return current;
395
396 const float decayed{ current * retained + target * (1.f - retained) };
397 return clamp(decayed, min(current, target), max(current, target));
398 }

423 inline float3 refract(const float3 i, const float3 n, float eta) noexcept
424 {
425 const float ndoti{ dot(i, n) };
426 float k{ 1.f - eta * eta * (1.f - ndoti * ndoti) };
427
428 if (k < 0.f) return float3_zero; // total internal reflection
429
430 return eta * i - (eta * dot(i, n) + sqrt(k)) * n;
